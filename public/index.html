const express = require('express'); const qrcode = require('qrcode'); const qrcodeTerminal = require('qrcode-terminal'); const { Client, LocalAuth } = require('whatsapp-web.js'); // --- SERVIDOR WEB PRA RENDER NÃO DESLIGAR --- const app = express(); let lastQR = null; app.get('/', (req, res) => { res.send('Bot Marmitaria ON! Acesse /qr para escanear'); }); app.get('/qr', async (req, res) => { if (!lastQR) return res.send('QR ainda não gerado ou já conectado. Veja os Logs no Render.'); try { const qrImage = await qrcode.toDataURL(lastQR); res.send(`

Escaneie no WhatsApp:


Atualize a página para novo QR

`); } catch (e) { res.send('Erro ao gerar QR'); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log('Servidor web rodando na porta ' + PORT)); // --- BOT WHATSAPP --- const client = new Client({ authStrategy: new LocalAuth({ dataPath: './.wwebjs_auth' }), puppeteer: { headless: true, executablePath: process.env.PUPPETEER_EXECUTABLE_PATH || undefined, args: ['--no-sandbox', '--disable-setuid-sandbox', '--disable-dev-shm-usage', '--disable-gpu', '--single-process', '--no-zygote'] } }); client.on('qr', qr => { console.log('QR GERADO, abra /qr para escanear'); qrcodeTerminal.generate(qr, { small: true }); lastQR = qr; }); client.on('ready', () => { console.log('Bot da Marmitaria ON!'); lastQR = null; }); const pedidos = {}; const DONO = (process.env.DONO_NUMERO || '5511999999999') + '@c.us'; // coloca seu numero no Render const CARDAPIO = `*🍱 MARMITARIA - CARDÁPIO*\nDigite o número:\n\n1 - Frango Grelhado R$18\n2 - Carne Assada R$20\n3 - Feijoada R$22\n4 - Strogonoff R$22\n\nDepois digite FINALIZAR`; client.on('message', async msg => { if (msg.from === 'status@broadcast') return; if (msg.from.includes('@g.us')) return; console.log('CHEGOU:', msg.body); const numero = msg.from; const texto = msg.body.trim().toUpperCase(); if (['OI','OLA','MENU','CARDAPIO'].includes(texto)) { pedidos[numero] = { itens: [], etapa: 'escolhendo' }; await msg.reply(`Olá! Bem-vindo(a)! 😊\n\n${CARDAPIO}`); return; } if (!pedidos[numero]) { await msg.reply('Digite OI para ver o cardápio 🍱'); return; } let pedido = pedidos[numero]; if (pedido.etapa === 'escolhendo') { if (['1','2','3','4'].includes(texto)) { const nomes = {'1':'Frango Grelhado 18','2':'Carne Assada 20','3':'Feijoada 22','4':'Strogonoff 22'}; const valores = {'1':18,'2':20,'3':22,'4':22}; pedido.itens.push({ nome: nomes[texto], valor: valores[texto] }); await msg.reply(`Adicionado: ${nomes[texto]} ✅\n\nDigite outro número ou FINALIZAR`); } else if (texto === 'FINALIZAR') { if (pedido.itens.length === 0) { await msg.reply('Carrinho vazio. Escolha 1 a 4.'); return; } pedido.etapa = 'endereco'; await msg.reply('Agora manda seu ENDEREÇO completo, ou digite RETIRADA'); } else { await msg.reply(CARDAPIO); } return; } if (pedido.etapa === 'endereco') { pedido.endereco = msg.body; pedido.etapa = 'pagamento'; await msg.reply('Endereço anotado! ✅\n\nPagamento?\n1 - PIX\n2 - Dinheiro\n3 - Cartão'); return; } if (pedido.etapa === 'pagamento') { let total = pedido.itens.reduce((s,i) => s + i.valor, 0); let lista = pedido.itens.map(i => `- ${i.nome}`).join('\n'); let pag = texto.includes('1') ? 'PIX' : texto.includes('2') ? 'Dinheiro' : 'Cartão'; let resumo = `*NOVO PEDIDO* 🛵\nCliente: ${numero}\n\n${lista}\nTotal: R$${total}\nEntrega: ${pedido.endereco}\nPagamento: ${pag}`; await client.sendMessage(DONO, resumo); await msg.reply(`Pedido FINALIZADO! 🎉\n\n${lista}\nTotal: R$${total}\n\nJá foi pra cozinha!`); delete pedidos[numero]; return; } }); client.initialize();

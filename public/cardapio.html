let carrinho = []; let categorias = []; let itens = []; let catAtiva = ''; // Carregar dados do backend async function carregarDados() { try { const [catRes, itRes] = await Promise.all([ fetch('/api/menu/categories').then(r => r.json()), fetch('/api/menu/items').then(r => r.json()) ]); categorias = catRes; itens = itRes; renderCategorias(); renderItens(); } catch (e) { document.getElementById('items').innerHTML = `
⚠️ Erro ao carregar. Atualize a página.
`; } } // Renderizar categorias function renderCategorias() { const container = document.getElementById('categories'); container.innerHTML = ''; const btnTodos = document.createElement('button'); btnTodos.className = 'cat-btn' + (catAtiva === '' ? ' active' : ''); btnTodos.textContent = 'Todos'; btnTodos.dataset.cat = ''; btnTodos.onclick = () => { catAtiva = ''; renderCategorias(); renderItens(); }; container.appendChild(btnTodos); categorias.forEach(cat => { const btn = document.createElement('button'); btn.className = 'cat-btn' + (catAtiva === cat.id ? ' active' : ''); btn.textContent = cat.name; btn.dataset.cat = cat.id; btn.onclick = () => { catAtiva = cat.id; renderCategorias(); renderItens(); }; container.appendChild(btn); }); } // Renderizar itens filtrados function renderItens() { const container = document.getElementById('items'); container.innerHTML = ''; const itensFiltrados = catAtiva ? itens.filter(item => item.categoryId === catAtiva) : itens; if (itensFiltrados.length === 0) { container.innerHTML = '

Nenhum item encontrado.

'; return; } itensFiltrados.forEach(item => { const card = document.createElement('div'); card.className = 'item-card'; card.innerHTML = `

${item.name}

${item.description || ''}

R$ ${item.price.toFixed(2).replace('.', ',')}
`; container.appendChild(card); }); } // Adicionar item ao carrinho function adicionarAoCarrinho(itemId) { const item = itens.find(i => i.id === itemId); if (!item) return; const existente = carrinho.find(i => i.id === itemId); if (existente) { existente.quantidade++; } else { carrinho.push({ ...item, quantidade: 1 }); } atualizarCarrinho(); } // Atualizar carrinho e interface function atualizarCarrinho() { const cartBar = document.getElementById('cartBar'); const cartCount = document.getElementById('cartCount'); const cartTotal = document.getElementById('cartTotal'); if (carrinho.length === 0) { cartBar.style.display = 'none'; return; } cartBar.style.display = 'flex'; let totalItens = 0; let totalPreco = 0; carrinho.forEach(item => { totalItens += item.quantidade; totalPreco += item.price * item.quantidade; }); cartCount.textContent = `${totalItens} item${totalItens > 1 ? 's' : ''}`; cartTotal.textContent = `R$ ${totalPreco.toFixed(2).replace('.', ',')}`; } // Abrir modal de finalização function abrirModal() { if (carrinho.length === 0) return alert('Seu carrinho está vazio.'); const resumo = document.getElementById('resumoPedido'); resumo.innerHTML = ''; carrinho.forEach(item => { const div = document.createElement('div'); div.className = 'resumo-item'; div.innerHTML = ` ${item.name} x${item.quantidade} R$ ${(item.price * item.quantidade).toFixed(2).replace('.', ',')} `; resumo.appendChild(div); }); const total = carrinho.reduce((acc, item) => acc + item.price * item.quantidade, 0); const divTotal = document.createElement('div'); divTotal.className = 'resumo-total'; divTotal.textContent = `Total: R$ ${total.toFixed(2).replace('.', ',')}`; resumo.appendChild(divTotal); document.getElementById('checkoutModal').classList.add('open'); } // Fechar modal function fecharModal() { document.getElementById('checkoutModal').classList.remove('open'); } // Enviar pedido para backend async function enviarPedido() { const nome = document.getElementById('nome').value.trim(); const telefone = document.getElementById('telefone').value.trim(); const endereco = document.getElementById('endereco').value.trim(); const pagamento = document.getElementById('pagamento').value; const observacoes = document.getElementById('observacoes').value.trim(); if (!nome || !telefone || !pagamento) { alert('Por favor, preencha os campos obrigatórios: nome, telefone e forma de pagamento.'); return; } const pedido = { cliente: { nome, telefone, endereco }, pagamento, observacoes, itens: carrinho.map(({ id, quantidade }) => ({ id, quantidade })), total: carrinho.reduce((acc, item) => acc + item.price * item.quantidade, 0) }; try { const res = await fetch('/api/orders', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(pedido) }); if (!res.ok) throw new Error('Erro ao enviar pedido.'); alert('Pedido enviado com sucesso! Obrigado pela preferência.'); carrinho = []; atualizarCarrinho(); fecharModal(); // Limpar campos document.getElementById('nome').value = ''; document.getElementById('telefone').value = ''; document.getElementById('endereco').value = ''; document.getElementById('pagamento').value = ''; document.getElementById('observacoes').value = ''; } catch (e) { alert('Erro ao enviar pedido. Tente novamente.'); console.error(e); } } // Inicializar carregarDados();
